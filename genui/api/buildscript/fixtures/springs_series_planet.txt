@@PREDICT
On a planet with weaker gravity, will the same mass stretch the two springs more or less than on Earth?
- More
- Less
- The same
Check: drop a mass on the hanger, then drag the gravity marker along the planet scale and watch the stretch.
@@IDEA
Two springs in series feel the same pull, so together they are softer than either one.
The combined stiffness is k_total = 1 / (1 / k1 + 1 / k2), and the stretch is x = W / k_total, where W = m g.
@@STEPS
- Drop a mass on the hanger and read the total stretch.
- Drag the gravity marker to a weaker planet and read it again.
- Drag the ring of the lower spring to make it stiffer and watch the total stretch fall.

var svg = document.getElementById("bench-art");
function pt(e) {
  var r = svg.getBoundingClientRect(), v = svg.viewBox.baseVal;
  return { x: v.x + (e.clientX - r.left) / r.width * v.width, y: v.y + (e.clientY - r.top) / r.height * v.height };
}
function track(handle, id, axis, a, b, lo, hi) {
  var held = false;
  handle.addEventListener("pointerdown", function (e) { held = true; handle.setPointerCapture(e.pointerId); });
  handle.addEventListener("pointermove", function (e) {
    if (!held) return;
    var f = Math.min(1, Math.max(0, (pt(e)[axis] - a) / (b - a)));
    host.set(id, lo + f * (hi - lo));
  });
  handle.addEventListener("pointerup", function () { held = false; });
  handle.addEventListener("pointercancel", function () { held = false; });
  host.on(id, function (v) { handle.setAttribute(axis === "x" ? "cx" : "cy", String(a + (v - lo) / (hi - lo) * (b - a))); });
}
function drop(handle, id, target, step, lo, hi, home) {
  var held = false;
  handle.addEventListener("pointerdown", function (e) { held = true; handle.setPointerCapture(e.pointerId); });
  handle.addEventListener("pointermove", function (e) {
    if (!held) return;
    var p = pt(e);
    handle.setAttribute("transform", "translate(" + (p.x - home.x) + " " + (p.y - home.y) + ")");
  });
  handle.addEventListener("pointerup", function (e) {
    if (!held) return;
    held = false;
    handle.removeAttribute("transform");
    var t = document.getElementById(target).getBoundingClientRect();
    if (e.clientX >= t.left && e.clientX <= t.right && e.clientY >= t.top && e.clientY <= t.bottom) {
      var next = host.value(id) + step;
      host.set(id, next > hi + 1e-9 ? lo : next);
    }
  });
  handle.addEventListener("pointercancel", function () { held = false; handle.removeAttribute("transform"); });
}
track(document.getElementById("ring1"), "k1", "y", 60, 230, 10, 50);
track(document.getElementById("ring2"), "k2", "y", 60, 230, 10, 50);
track(document.getElementById("gmark"), "g_planet", "x", 120, 296, 1, 25);

var s1 = document.getElementById("spring1"), s2 = document.getElementById("spring2");
var load = document.getElementById("load"), hanger = document.getElementById("hanger");
function coil(x, y0, y1) {
  var pts = [x + "," + y0], n = 8;
  for (var i = 0; i < n; i++) pts.push((x + (i % 2 ? -14 : 14)) + "," + (y0 + (i + 0.5) * (y1 - y0) / n));
  pts.push(x + "," + y1);
  return pts.join(" ");
}
function draw() {
  var x1 = host.value("x1"), x2 = host.value("x2"), m = host.value("mass");
  var e1 = 24 + 60 + x1 * 45, e2 = e1 + 12 + 60 + x2 * 45;
  s1.setAttribute("points", coil(130, 24, e1));
  s2.setAttribute("points", coil(130, e1 + 12, e2));
  var h = 6 + m * 60;
  load.setAttribute("y", String(e2 + 4));
  load.setAttribute("height", String(h));
  hanger.setAttribute("y", String(e2 - 10));
  svg.setAttribute("aria-label", "Two springs stretched by " + (host.value("x") * 100).toFixed(1) + " centimetres in total");
}
host.on("x1", draw);
host.on("x2", draw);
host.on("mass", draw);
host.on("x_cm", function (v) { host.found("Together the two springs stretch " + v.toFixed(1) + " cm."); });
@@HINTS
- Both springs carry the whole weight, so each one stretches as if it hung alone.
- A weaker planet pulls the same mass with less force.
@@EXPLAIN
Each spring in the row feels the whole weight, and each stretches by the weight divided by its own stiffness. The two stretches add, so the pair behaves like one softer spring.

A planet with weaker gravity pulls the same mass with a smaller weight, so every stretch gets smaller in the same proportion.
