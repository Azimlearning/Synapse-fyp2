A pendulum on another planet
@@KICKER
Physics, introductory
@@TRY
Drag the ring on the right-hand rail down to lengthen the pendulum and watch the swing slow down.
@@PREDICT
On a planet with weaker gravity, will the same pendulum swing faster or slower than on Earth?
- Faster
- Slower
- At the same speed
Check: drag the gravity ring along its rail to a weaker value and compare the time for one swing.
@@IDEA
For small swings the time for one swing depends on the length and on gravity, not on the mass or on how far you pull it back.
T = 2 pi sqrt(L / g). The speed at the bottom comes from the height the bob falls: v = sqrt(2 g L (1 - cos angle)).
@@STEPS
- Drag the length ring down to make the string longer and read the time for one swing.
- Drag the gravity ring to a weaker planet and read it again.
- Drag the angle ring further out and notice that the time stays put while the speed at the bottom grows.

@@SCRIPT
var svg = document.getElementById("bench-art");
function pt(e) {
  var r = svg.getBoundingClientRect(), v = svg.viewBox.baseVal;
  return { x: v.x + (e.clientX - r.left) / r.width * v.width, y: v.y + (e.clientY - r.top) / r.height * v.height };
}
function track(handle, id, axis, a, b, lo, hi) {
  var held = false;
  handle.addEventListener("pointerdown", function (e) { held = true; handle.setPointerCapture(e.pointerId); });
  handle.addEventListener("pointermove", function (e) {
    if (!held) return;
    var f = Math.min(1, Math.max(0, (pt(e)[axis] - a) / (b - a)));
    host.set(id, lo + f * (hi - lo));
  });
  handle.addEventListener("pointerup", function () { held = false; });
  handle.addEventListener("pointercancel", function () { held = false; });
  host.on(id, function (v) { handle.setAttribute(axis === "x" ? "cx" : "cy", String(a + (v - lo) / (hi - lo) * (b - a))); });
}
track(document.getElementById("lring"), "length", "y", 50, 270, 0.3, 1.5);
track(document.getElementById("aring"), "start_angle", "x", 40, 280, 5, 40);
track(document.getElementById("gring"), "g_planet", "x", 40, 280, 1, 25);

var swing = document.getElementById("swing"), rod = document.getElementById("rod"), bob = document.getElementById("bob");
function pose(deg) { swing.setAttribute("transform", "rotate(" + deg + " 160 24)"); }
host.on("length", function (len) {
  var y = 24 + len * 100;
  rod.setAttribute("y2", String(y));
  bob.setAttribute("cy", String(y));
});
host.on("period", function (t) {
  svg.setAttribute("aria-label", "A pendulum taking " + t.toFixed(2) + " seconds for one swing");
  host.found("One swing takes " + t.toFixed(2) + " s.");
});
if (host.reducedMotion) {
  host.on("start_angle", pose);
} else {
  var t0 = null;
  var spin = function (now) {
    if (t0 === null) t0 = now;
    var T = host.value("period"), A = host.value("start_angle");
    pose(A * Math.cos(2 * Math.PI * ((now - t0) / 1000) / T));
    requestAnimationFrame(spin);
  };
  requestAnimationFrame(spin);
}
@@HINTS
- A longer string means a longer path for the bob to travel.
- Weaker gravity pulls the bob back more gently, so each swing takes longer.
@@EXPLAIN
The bob is pulled back toward the middle by gravity. A longer string makes the path longer and the pull along it weaker, so one swing takes more time. Weaker gravity weakens the pull in the same way.

Pulling the bob further out gives it more height to fall, so it moves faster at the bottom. For small angles it also has further to go, and the two effects cancel, which is why the time hardly changes.
